cmake_minimum_required(VERSION 3.18)
project(vitqra_speech LANGUAGES C)
set(SHERPA_SOURCE "" CACHE PATH "Audited sherpa-onnx source root containing sherpa-onnx/c-api/c-api.h")
set(SHERPA_LIBRARY "" CACHE FILEPATH "Matching reviewed sherpa-onnx C API shared library")
if(NOT EXISTS "${SHERPA_SOURCE}/sherpa-onnx/c-api/c-api.h" OR NOT EXISTS "${SHERPA_LIBRARY}")
    message(FATAL_ERROR "Supply matching reviewed sherpa source and library")
endif()
if(ANDROID AND NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "VITQRA Quest targets arm64-v8a")
endif()
add_library(sherpa_c_api SHARED IMPORTED)
set_target_properties(sherpa_c_api PROPERTIES IMPORTED_LOCATION "${SHERPA_LIBRARY}")
add_library(vitqra_speech SHARED vitqra_speech.c)
target_include_directories(vitqra_speech PRIVATE "${SHERPA_SOURCE}")
target_link_libraries(vitqra_speech PRIVATE sherpa_c_api)
set_target_properties(vitqra_speech PROPERTIES C_STANDARD 99 C_VISIBILITY_PRESET hidden)
