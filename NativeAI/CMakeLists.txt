cmake_minimum_required(VERSION 3.22)
project(vitqra_llm LANGUAGES C CXX)
set(LLAMA_SOURCE "" CACHE PATH "Reviewed clean llama.cpp source checkout")
set(VITQRA_LLAMA_PIN "9c003a97ab34a69cc49243f9b02014038205ffae")
if(NOT EXISTS "${LLAMA_SOURCE}/include/llama.h")
    message(FATAL_ERROR "Supply the reviewed llama.cpp source")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${LLAMA_SOURCE}" rev-parse HEAD
    OUTPUT_VARIABLE ACTUAL_PIN OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE PIN_STATUS)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${LLAMA_SOURCE}" status --porcelain --untracked-files=all
    OUTPUT_VARIABLE DIRTY OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE STATUS_CODE)
if(NOT PIN_STATUS EQUAL 0 OR NOT STATUS_CODE EQUAL 0 OR NOT ACTUAL_PIN STREQUAL VITQRA_LLAMA_PIN OR NOT DIRTY STREQUAL "")
    message(FATAL_ERROR "Expected exact clean reviewed upstream revision")
endif()
if(ANDROID AND NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "Quest build requires arm64-v8a")
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(OPTION LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_TOOLS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_SERVER LLAMA_BUILD_APP LLAMA_BUILD_UI LLAMA_BUILD_MTMD LLAMA_BUILD_IS_DEV GGML_RPC GGML_BACKEND_DL GGML_CUDA GGML_HIP GGML_VULKAN GGML_METAL GGML_OPENCL GGML_SYCL GGML_BLAS GGML_OPENMP GGML_OPENMP_FETCH GGML_CPU_KLEIDIAI GGML_NATIVE)
    set(${OPTION} OFF CACHE BOOL "" FORCE)
endforeach()
set(GGML_CPU ON CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE}" llama-build EXCLUDE_FROM_ALL)
add_library(vitqra_llm SHARED vitqra_llm.cpp)
target_link_libraries(vitqra_llm PRIVATE llama)
target_compile_features(vitqra_llm PRIVATE cxx_std_17)
set_target_properties(vitqra_llm PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
